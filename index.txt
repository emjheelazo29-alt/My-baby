  <script>
    // ---- Floating hearts generator ----
    (function hearts(){
      const container = document.querySelector('.hearts');
      const colors = ['#ff6fb5','#ffd166','#9b5de5','#f15bb5','#f79d65'];
      const makeHeart = () => {
        const h = document.createElement('div');
        h.className = 'heart';
        const size = Math.random() * 14 + 10; // 10–24px
        h.style.width = size+'px';
        h.style.height = size+'px';
        h.style.left = Math.random()*100+'vw';
        h.style.bottom = '-10vh';
        h.style.animationDuration = (Math.random()*6 + 8) + 's';
        h.style.animationDelay = (Math.random()*4) + 's';
        h.style.opacity = (Math.random()*0.5 + 0.35).toFixed(2);
        const c = colors[Math.floor(Math.random()*colors.length)];
        h.style.setProperty('--accent', c);
        container.appendChild(h);
        setTimeout(()=> h.remove(), 16000);
      };
      setInterval(makeHeart, 600);
      for(let i=0;i<18;i++) makeHeart();
    })();

    // ---- Countdown to Dec 9 (local time) ----
    (function countdown(){
      const banner = document.getElementById('countdownBanner');
      const text = document.getElementById('countdownText');
      const now = new Date();
      const year = now.getFullYear();
      const target = new Date(year, 11, 9, 0, 0, 0); // December=11
      const update = () => {
        const n = new Date();
        let t = target;
        if(n > target){
          // If past Dec 9 this year, switch message
          banner.classList.add('small');
          text.textContent = "Happy Birthday! Today is for you — and always will be.";
          return;
        }
        const diff = t - n;
        const d = Math.floor(diff / (1000*60*60*24));
        const h = Math.floor(diff / (1000*60*60)) % 24;
        const m = Math.floor(diff / (1000*60)) % 60;
        const s = Math.floor(diff / 1000) % 60;
        text.textContent = `Countdown: ${d}d ${h}h ${m}m ${s}s until Dec 9`;
        requestAnimationFrame(update);
      };
      update();
    })();

    // ---- Typewriter reveal message ----
    const revealBtn = document.getElementById('revealBtn');
    const letterEl  = document.getElementById('letter');
    const typedEl   = document.getElementById('typed');

    const message = `
To the person who turns ordinary days into my favorite memories:

Happy birthday. On Dec 9 and every day after, I choose you.
Thank you for your kindness, your laugh, and the way you make me braver.

I can’t promise perfect paths, but I can promise honest steps  with you.
Let's keep writing our story, slowly and truly, one moment at a time. Baby nako i loveee youuuuu sooo muchhh poooo you deserve everything.

— your husby hihih
`;

    function typewriter(text, el, speed = 22){
      el.textContent = '';
      let i = 0;
      const tick = () => {
        if(i <= text.length){
          el.textContent = text.slice(0, i++);
          setTimeout(tick, speed);
        }
      };
      tick();
    }

    revealBtn.addEventListener('click', () => {
      letterEl.classList.add('show');
      typewriter(message, typedEl, 18);
      revealBtn.disabled = true;
      revealBtn.textContent = '💖 Message revealed';
      // subtle burst of hearts
      for(let i=0;i<12;i++){
        setTimeout(()=> {
          const evt = new Event('spawnHeart');
          document.dispatchEvent(evt);
        }, i*90);
      }
    });

    // ---- Optional music play (requires user gesture) ----
    const audio = document.getElementById('audio');
    const musicBtn = document.getElementById('musicBtn');
    musicBtn.addEventListener('click', async () => {
      try{
        if(audio.src){
          if(audio.paused){
            await audio.play();
            musicBtn.textContent = 'Pause our song';
          }else{
            audio.pause();
            musicBtn.textContent = 'Play our song (optional)';
          }
        }else{
          alert('Add your song file first: set the <source> src in the HTML.');
        }
      }catch(e){
        alert('Tap again to allow audio — some browsers need a user gesture.');
      }
    });

    // allow extra heart spawns
    document.addEventListener('spawnHeart', () => {
      const container = document.querySelector('.hearts');
      const h = document.createElement('div');
      h.className = 'heart';
      h.style.left = Math.random()*100+'vw';
      h.style.bottom = '-8vh';
      h.style.animationDuration = (Math.random()*6 + 6) + 's';
      container.appendChild(h);
      setTimeout(()=> h.remove(), 14000);
    });
